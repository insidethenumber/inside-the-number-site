           The same endpoint also returns Dana White's Contender Series, which

           prices. Filtered here to match index.html and ufc.html. */
        const NOT_UFC=/contender series|dana white/i;
        D[label]=(d.events||[]).filter(ev=>!NOT_UFC.test(ev.name||ev.shortName||''))

          const cs=c.competitors||[]; if(cs.length<2) return null;
          const [x,y]=cs;
          const nm=z=>z.athlete?.shortName||z.athlete?.displayName||'';
          return{id:c.id,league:label,kind:'mma',state:ev.status?.type?.state,
            detail:c.status?.type?.shortDetail||ev.status?.type?.shortDetail||'',
            card:ev.name||'',weight:c.type?.abbreviation||'',venue:c.venue?.fullName||'',

               rec:y.records?.find(r=>r.type==='total')?.summary||''},
            /* Fights can carry a published pick like any other contest. This was
               hardcoded to null, so a UFC play written into todaysGames could
               never surface on the fight page. Matched on last names, and tried
               both orders because there is no home or away in a fight — whoever
               wrote the pick may have listed them the other way round. */
            o:c.odds?.[0]||{},
            ours:OURS[mmaKey(nm(x),nm(y))]||OURS[mmaKey(nm(y),nm(x))]||null};
        })).filter(Boolean);
